# shennong - QCOM FBE 解密所需 vendor blobs (从 Android 17 原厂包提取)
# 用法: ./extract-files.sh <HyperOS A17 的 system/vendor 镜像目录 或 adb>

# 注意: A17 上 KeyMint/Weaver 的 NDK 版本可能从 V3/V2 升级到 V4,
# extract 时若提示某条缺失, 请用下面命令从真机查找实际文件名后替换:
#   adb shell "ls /vendor/lib64/android.hardware.security.* /vendor/lib64/android.hardware.gatekeeper* /vendor/lib64/android.hardware.weaver*"
